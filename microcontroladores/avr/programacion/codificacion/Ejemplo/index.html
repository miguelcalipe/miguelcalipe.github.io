2. DESCRIPCIÓN DEL PROGRAMA

 
 /**
 * ============================================================================
 * DESCRIPCIÓN
 * ============================================================================
 *
 * Programa desarrollado en lenguaje C para un microcontrolador AVR de 8 bits.
 *
 * El programa configura el pin PB1 (Bit 1 del puerto B) como pin de salida 
 * para el indicador LED, definido como LED_PIN
 * Configura el pin PB0 (Bit 0 del puerto B) como pin de entrada 
 * para el pulsador definido como BUTTON_PIN
 * 
 * Mientras el pulsador esté en reposo, el led permanece apagado. 
 * Cuando se presiona el pulsador, el LED se enciende.
 * 
 * El programa utiliza operadores OR bitwise, AND bitwise para establecer la entrada y salida
 * También utiliza una compuerta AND invertida como comparador para detectar si un botón 
 * está presionado (activo en bajo). 
 *
 * ============================================================================
 * OBJETIVOS
 * ============================================================================
 * 1. Configurar un puerto GPIO del microcontrolador AVR como salida y entrada.
 * 2. Comprender el funcionamiento y la manipulación del registro DDRB.
 * 3. Activar y utilizar la resistencia Pull-Up interna mediante el registro PORTB.
 * 4. Manipular bits específicos del registro PORTB de forma no destructiva 
 *    utilizando operadores lógicos (OR |=, AND &=) y de desplazamiento (<<, ~).
 * 5. Implementar retardos mediante _delay_ms().
 * 6. Leer e inspeccionar el estado de entradas digitales en el registro PINB.
 * 7. Implementar un filtrado anti-rebote (software debounce) mediante _delay_ms().
 * 8. Estructurar un bucle principal de ejecución continua (while(1)).
 * 
 * ============================================================================
 * MICROCONTROLADOR Y HERRAMIENTAS
 * ============================================================================
 *
 * Microcontrolador : AVR de 8 bits
 * Dispositivo      : ATmega88PA / ATmega328P
 * Compilador       : avr-gcc
 * Lenguaje         : C
 *
 * El dispositivo concreto depende del microcontrolador seleccionado durante
 * la compilación.
 *
 * ============================================================================
 * PERIFÉRICOS UTILIZADOS
 * ============================================================================
 * Puerto utilizado : PORTB
 * Pines utilizados  : PB0 y PB1
 * Funciones         : 
 *   - PB0 (BUTTON_PIN) : Entrada digital para lectura de pulsador físico
 *                        con resistencia Pull-Up interna activa.
 *   - PB1 (LED_PIN)    : Salida digital para control de actuador/LED.
 * 
 */

3. PRINCIPIOS DE FUNCIONAMIENTO


 
/* ============================================================================
 * PRINCIPIO DE FUNCIONAMIENTO
 * ============================================================================
 *
 * 1. DIRECCIÓN DE DATOS (DDRB):
 *    El registro DDRB determina la dirección de los pines del Puerto B:
 *        DDRB bit = 0  → Entrada
 *        DDRB bit = 1  → Salida
 *
 *    - DDRB |= (1 << PB1)         : Configura PB1 como salida sin borrar los
 *                                   demás bits (bit-wise OR).
 *    - DDRB &= ~(1 << PB0)        : Configura PB0 como entrada limpiando 
 *                                   el bit 0 mediante máscara invertida.
 *
 * 2. RESISTENCIA PULL-UP INTERNA (PORTB):
 *    Al escribir un '1' en el registro PORTB sobre un pin configurado como 
 *    entrada, se activa su resistencia Pull-Up interna, manteniendo el pin 
 *    en nivel lógico alto (1 / 5V) cuando no se presiona el pulsador:
 *        PORTB |= (1 << PB0);
 *
 *    Esto mantiene a PB0 en nivel ALTO (VCC / 5V) en reposo. Cuando el 
 *    pulsador se presiona, conecta el pin directamente a Tierra (GND / 0V).
 *
 * 3. LECTURA Y ENMASCARAMIENTO (PINB):
 *    Para consultar el estado real del pin se examina el registro PINB:
 *        PINB & (1 << PB0)
 *
 *    Esta expresión aísla el bit 0. La condición !(PINB & (1 << PB0)) evalúa 
 *    como VERDADERA únicamente cuando el botón está presionado (nivel 0V)[cite: 3].
 *
 * ============================================================================
 * TEMPORIZACIÓN Y ANTI-REBOTE (DEBOUNCE)
 * ============================================================================
 *
 * Tiempo de espera : 20 ms
 *
 * La temporización se implementa mediante _delay_ms(20).
 *
 * Los pulsadores mecánicos generan micro-vibraciones eléctricas al cerrarse.
 * El retardo de 20 ms introduce una ventana de estabilización bloqueante:
 * el procesador suspende momentáneamente la ejecución para ignorar el ruido 
 * mecánico inicial antes de reconfirmar mediante un segundo 'if' que la 
 * pulsación sigue siendo válida.
 *
 * ============================================================================
 * OBSERVACIONES
 * ============================================================================
 *
 * 1. DEFINICIÓN DE F_CPU:
 *    La macro F_CPU debe definirse (16000000UL) antes de incluir la cabecera
 *     para que los retardos por software sean calculados 
 *    con precisión de tiempo real[cite: 1].
 *
 * 2. MODIFICACIÓN NO DESTRUCTIVA:
 *    El uso estricto de operadores |= y &= evita la "destrucción de estado",
 *    permitiendo modificar únicamente los pines PB0 y PB1 sin alterar la 
 *    configuración previa de los pines PB2 a PB7[cite: 3].
 *
 * ============================================================================
 * REFERENCIAS DE PROGRAMACIÓN AVR
 * ============================================================================
 *
 * - avr/io.h    : Definición mnemónica de registros (DDRB, PORTB, PINB)[cite: 1, 3].
 * - util/delay.h: Funciones de retardo por software (_delay_ms)[cite: 1].
 *
 * ============================================================================
 */

4. CÓDIGO DEL PROGRAMA


/==============================================================================
 * ARCHIVO:       main.c
 * DISPOSITIVO:   Microcontrolador AVR de 8 bits (ej. ATmega328P / ATmega168)
 * FRECUENCIA:    16.0 MHz (F_CPU)
 * DESCRIPCIÓN:   Programa de control GPIO para la manipulación de registros 
 *                de hardware (SFR) utilizando técnicas de enmascaramiento 
 *                y filtrado de rebotes mecánico (debounce) en una entrada.
 
 ==============================================================================*/

/* ----------------------------------------------------------------------------
 * 1. DEFINICIÓN DE PARÁMETROS Y CABECERAS DEL SISTEMA
 * ---------------------------------------------------------------------------- */ 
#ifndef F_CPU
#define F_CPU 16000000UL // Define la frecuencia de reloj del chip (16 MHz)
#endif

#include <avr/io.h>  // Contiene las definiciones mnemónicas de registros (DDRB, PORTB, PINB)
#include <util/delay.h> // Proporciona funciones de retardo por software (_delay_ms)

/* ----------------------------------------------------------------------------
 * 2. DEFINICIÓN DE CONSTANTES SIMBÓLICAS Y PINES (MAPEO DE HARDWARE)
 * ---------------------------------------------------------------------------- */
#define LED_PIN     PB1 // Pin de salida para el indicador LED (Bit 1 del puerto B)
#define BUTTON_PIN  PB0 // Pin de entrada para el pulsador (Bit 0 del puerto B)

/* ----------------------------------------------------------------------------
 * 3. FUNCIÓN PRINCIPAL (ENTRY POINT)
 * ---------------------------------------------------------------------------- */
int main(void) {
   /* ========================================================================
     * SECCIÓN A: CONFIGURACIÓN E INICIALIZACIÓN DE PUERTOS (GPIO)
     * ======================================================================== */

    // 1. Configurar LED_PIN (PB1) como SALIDA (Output) mediante Bitwise OR (no destructivo). 
    // OR  (|)  --> Pone en 1, los bits específicos sin alterar el resto (SET / ENCENDER).
    // Cualquier bit que se opere con el bit 1 mediante OR, dará como resultado '1' en el bit  de salida.
    // Se utiliza el operador OR a nivel de bits (|=) para no alterar los demás pines de PORTB .
    // DDRB = DDRB | (1 << LED_PIN); // Configura PB1 como salida. Mas simplificadamente: 
    DDRB |= (1 << LED_PIN);
    
    // 2. Configurar BUTTON_PIN (PB0) como ENTRADA (Input) mediante Bitwise AND con máscara invertida
    // AND (&)  --> Pone en 0 (CLEAR) o evalúa el estado de bits específicos
    // Cualquier bit que se opere con el bit 0 mediante AND, dará como resultado '0' en el bit de salida.
    // Aquí se utiliza AND (&) con la máscara invertida (~) para colocar un '0' seguro en el bit 0 
    // sin alterar los demás pines de PORTB. Esto asegura que PB0 se configure como entrada.
    // DDRB = DDRB & ~(1 << BUTTON_PIN); // Configura PB0 como entrada. Mas simplificadamente:
    DDRB &= ~(1 << BUTTON_PIN);
    
    // 3. Habilitar la Resistencia Pull-Up Interna en BUTTON_PIN (PB0)
    // Al escribir un '1' en el registro PORTx cuando el pin está configurado como entrada, 
    // se activa la resistencia interna Pull-Up conectada a VCC
    // Al activarla, se mantiene el pin en un nivel lógico alto (1/ 5V) cuando el pulsador no está presionado.
    // Esto evita que el pin quede flotante y susceptible a interferencias o ruido eléctrico.
    // PORTB = PORTB | (1 << BUTTON_PIN); // Habilita Pull-Up en PB0. Mas simplificadamente:
    PORTB |= (1 << BUTTON_PIN);

    /* ========================================================================
     * SECCIÓN B: BUCLE PRINCIPAL DE EJECUCIÓN CONTINUA (INFINITE LOOP)
     * ======================================================================== */
    while (1) {
        /* --------------------------------------------------------------------
         * PASO 1: Muestrear el estado lógico del pulsador leyendo el registro PINB
         * Dado que la resistencia Pull-Up está activa, el pin PB0 lee '1' en 
         * reposo y '0' cuando el pulsador se presiona conectándolo a Tierra (GND)
         * -------------------------------------------------------------------- */
        if (!(PINB & (1 << BUTTON_PIN))){
            
            /* ----------------------------------------------------------------
             * PASO 2: Filtrado Anti-rebote (Software Debounce)
             * Se espera un intervalo de 20 ms para superar la ventana de 
             * vibración o ruido mecánico de los contactos eléctricos.
             * ---------------------------------------------------------------- */
            _delay_ms(20);

            /* ----------------------------------------------------------------
             * PASO 3: Verificación de seguridad
             * Se reevalúa el registro PINB. Si el bit sigue en '0', se confirma 
             * que es una pulsación real y no un falso disparo por ruido.
             * ---------------------------------------------------------------- */
            if (!(PINB & (1 << BUTTON_PIN))) {
                // Al activar la resistencia pull-up del pin (PORTB |= (1 << BUTTON_PIN);), 
                // el hardware mantiene el pin en un nivel lógico alto (1 / 5 V) cuando el pulsador no está presionado.
                // Operación de evaluación: (PINB & (1 << BUTTON_PIN)).
                // Botón PRESIONADO: La máscara  devuelve 0 (en C, equivale a Falso / FALSE)
                // La negación (!) invierte el estado de la mascara (0) --> 1; Verdadero / TRUE
                // Entonces se ejecuta el código del bloque if:
                PORTB |= (1 << LED_PIN);
            }

        } else {
            // Botón SUELTO: La máscara lee 1 en el bit PB0 (BUTTON_PIN). No importan los otros pines.
            // La expresión (PINB & (1 << BUTTON_PIN)) devuelve un valor diferente de cero 
            // en C, esto equivale a Verdadero / TRUE.  
            // La negación (!) invierte el estado de la mascara (1) --> 0; Falso / FALSE
            // Entonces se ejecuta la parte del código en el bloque else

            /* ----------------------------------------------------------------
             * PASO 4: Estado de reposo
             * Si el botón no está presionado, se desactiva la salida (0V).
             * ---------------------------------------------------------------- */
            PORTB &= ~(1 << LED_PIN);
        }
    }

    return 0; // Estructura formal de retorno según el estándar C[cite: 1]
}
